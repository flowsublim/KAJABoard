    t.BOOTSTRAP_JSON = JSON.stringify(boot).replace(/</g, '\\u003c');

    return HtmlService.createHtmlOutput('<h2>Portal error</h2><pre>'+PORTAL_escape_(err && err.message || err)+'</pre>').setTitle('Portal Error');

function PORTAL_getLoginBootstrap_(params) {
  var email = PORTAL_userEmail_();
  var auth = PORTAL_securityCheck_(email, '');
  var state = !email ? 'EMAIL_EMPTY' : (auth.allowed ? 'LOGIN_READY' : 'FORBIDDEN');
  return {
    app: { code: PORTAL_CFG.MODULE_CODE, version: PORTAL_CFG.VERSION, title: 'ERP CV KIRAL' },
    state: state,
    email: email,
    auth: auth,
    passport: '',
    modules: [],
    heartbeat: PORTAL_readGlobalHeartbeat_(),
    sessionPolicy: {
      ttlMs: PORTAL_CFG.SESSION_TTL_MS,
      hours: Math.round(PORTAL_CFG.SESSION_TTL_MS / 3600000),
      previewExpiresAt: auth.allowed ? PORTAL_formatDateTime_(new Date(Date.now() + PORTAL_CFG.SESSION_TTL_MS)) : ''
    },
    now: PORTAL_formatDateTime_(new Date()),
    forceLogin: !!(params && (params.login || params.logout || params.relogin || params.action === 'global_logout'))
  };
}

function PORTAL_login() {
  var email = PORTAL_userEmail_();
  var auth = PORTAL_securityCheck_(email, '');
  if (!auth.allowed) {
    return { ok:false, state:'FORBIDDEN', auth:auth, reason:auth.reason || 'Akses ditolak' };
  }
  PORTAL_ensureLogoutColumn_();
  var stamp = Date.now();
  var token = PORTAL_makePassport_(auth.email, stamp);
  var hb = PORTAL_bumpHeartbeat_('Login ' + auth.email);
  return {
    ok:true,
    state:'LOGGED_IN',
    auth:auth,
    passport:token,
    modules:PORTAL_getModulesForUser_(auth, token),
    heartbeat:hb,
    portalUrl: PORTAL_getPortalUrl_(),
    session: PORTAL_sessionInfoFromToken_(auth.email, token),
    now:PORTAL_formatDateTime_(new Date())
  };
}

function PORTAL_refreshBootstrap(passport) {
  var email = PORTAL_userEmail_();
  var auth = PORTAL_securityCheck_(email, passport);
  if (!auth.allowed) return { ok:false, auth:auth, modules:[], heartbeat:PORTAL_readGlobalHeartbeat_(), portalUrl:PORTAL_getPortalUrl_() };
  return { ok:true, auth:auth, passport:passport, modules:PORTAL_getModulesForUser_(auth, passport), heartbeat:PORTAL_readGlobalHeartbeat_(), session:PORTAL_sessionInfoFromToken_(auth.email, passport), portalUrl:PORTAL_getPortalUrl_() };
}

function PORTAL_globalHeartbeat(passport, clientVersion) {
  var email = PORTAL_userEmail_();
  var auth = PORTAL_securityCheck_(email, passport);
  var hb = PORTAL_readGlobalHeartbeat_();

    reason: auth.reason,
    shouldLogout: !auth.allowed,
    user: { email: auth.email || email || '', name: auth.displayName || auth.email || email || '', role: auth.role || '', department: auth.department || '' },
    serverVersion: hb.version,
    updatedAt: hb.updatedAt,
    shouldRefresh: !!clientVersion && String(clientVersion) !== String(hb.version),
    session: auth.allowed ? PORTAL_sessionInfoFromToken_(auth.email, passport) : null,
    portalUrl: PORTAL_getPortalUrl_(),
    now: PORTAL_formatDateTime_(new Date())
  };
}

function PORTAL_securityCheck_(email, passport) {
  email = PORTAL_normEmail_(email);
  if (!email) return { allowed:false, reason:'EMAIL_KOSONG', email:'' };
  var user = PORTAL_findUser_(email);
  if (!user) return { allowed:false, reason:'USER_TIDAK_ADA_DI_MASTER_USER', email:email };
  if (!PORTAL_isActive_(PORTAL_pickValue_(user, ['Status','Status_Akun','Aktif']))) return { allowed:false, reason:'USER_NONAKTIF', email:email, status:PORTAL_pickValue_(user, ['Status','Status_Akun','Aktif']) || '' };

  if (passport) {
    var pv = PORTAL_validatePassport_(email, passport);
    if (!pv.ok) return { allowed:false, reason:'PASSPORT_' + pv.reason, email:email };
  }

  var role = PORTAL_pickValue_(user, ['Role','Jabatan','Hak_Akses','Akses']) || '';
  var department = PORTAL_pickValue_(user, ['Department','Departemen','Divisi']) || '';
  var allowedModules = PORTAL_pickValue_(user, ['Allowed_Modules','Allowed Modules','Module_Access','Hak_Modul','Akses_Modul','Modul']) || '';
  var isAdmin = PORTAL_key_(role).indexOf('ADMIN') !== -1 || PORTAL_key_(department).indexOf('ADMIN') !== -1 || PORTAL_key_(allowedModules).indexOf('SUPERADMIN') !== -1 || PORTAL_key_(allowedModules).indexOf('ALL') !== -1;
  return {
    allowed:true,
    reason:isAdmin ? 'ADMIN' : 'ACTIVE_USER',
    email:email,
    displayName:PORTAL_userDisplayName_(user, email),
    role:role,
    department:department,
    allowedModules:allowedModules,
    isAdmin:isAdmin,
    loginAt: passport ? PORTAL_parsePassport_(passport).stamp : 0
  };
}

function PORTAL_getModulesForUser_(auth, passport) {
  var sh = PORTAL_master_().getSheetByName(PORTAL_SHEETS.MASTER_MODULE);
  if (!sh) return [];
  var rows = PORTAL_readRows_(sh);
  var out = [];
  rows.forEach(function(r) {
    var code = PORTAL_clean_(r.Module_Code || r.Code || r.ID_Modul || r.Kode_Modul || '');
    var name = PORTAL_clean_(r.Module_Name || r.Name || r.Nama_Modul || r.Nama || code);
    var url = PORTAL_clean_(r.Web_App_URL || r.WebApp_URL || r.URL_Web_Aktif || r.URL || r.Link || '');
    if (!code || !url) return;
    if (!PORTAL_moduleStatusAllowed_(r.Status || r.Aktif)) return;
    if (PORTAL_key_(code) === 'PORTAL') return;
    if (!auth.isAdmin && !PORTAL_userCanOpenModule_(auth, code, name)) return;
    var sep = url.indexOf('?') === -1 ? '?' : '&';
    var withPassport = url + sep + 'vouch=' + encodeURIComponent(auth.email || '') + '&paspor=' + encodeURIComponent(passport || '') + '&passport=' + encodeURIComponent(passport || '') + '&from=PORTAL';
    out.push({ code:code, id:code, name:name, nama:name, url:withPassport, rawUrl:url, status:r.Status || '', notes:r.Notes || '', icon:PORTAL_iconForModule_(code, name) });
  });
  out.sort(function(a,b){ return PORTAL_moduleOrder_(a.code)-PORTAL_moduleOrder_(b.code) || a.name.localeCompare(b.name); });
  return out;
}

function PORTAL_userCanOpenModule_(auth, code, name) {
  var targets = [PORTAL_key_(code), PORTAL_key_(name)];
  var fields = [auth.allowedModules, auth.role, auth.department].map(PORTAL_key_).join('|');
  if (fields.indexOf('ALL') !== -1 || fields.indexOf('SUPERADMIN') !== -1) return true;
  return targets.some(function(t){ return t && fields.indexOf(t) !== -1; });
}

function PORTAL_makePassport_(email, stamp) {
  stamp = Number(stamp) || Date.now();
  return stamp + ':' + PORTAL_hashPassport_(email, stamp);
}

function PORTAL_parsePassport_(passport) {
  var raw = PORTAL_clean_(passport);
  var m = raw.match(/^(\d{10,}):([a-f0-9]{64})$/i);
  return m ? { stamp:Number(m[1]) || 0, hash:String(m[2] || '').toLowerCase() } : { stamp:0, hash:raw.toLowerCase() };
}

function PORTAL_validatePassport_(email, passport) {
  var p = PORTAL_parsePassport_(passport);
  if (!p.stamp || !p.hash) return { ok:false, reason:'FORMAT_TIDAK_VALID' };
  if (p.hash !== PORTAL_hashPassport_(email, p.stamp)) return { ok:false, reason:'HASH_TIDAK_VALID' };
  if (Date.now() - p.stamp > PORTAL_CFG.SESSION_TTL_MS) return { ok:false, reason:'EXPIRED' };
  var lastLogout = PORTAL_getLastLogoutStamp_(email);
  if (lastLogout && p.stamp < lastLogout) return { ok:false, reason:'GLOBAL_LOGOUT' };
  return { ok:true, stamp:p.stamp };
}

function PORTAL_hashPassport_(email, stamp) {
  var raw = PORTAL_normEmail_(email) + '|' + Number(stamp || 0) + '|' + PORTAL_secret_();
  return Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, raw)
    .map(function(b){ return (b < 0 ? b + 256 : b).toString(16).padStart(2, '0'); })
    .join('');
}

function PORTAL_secret_() { return PORTAL_CFG.SHARED_SECRET; }

function PORTAL_sessionInfoFromToken_(email, passport) {
  var p = PORTAL_parsePassport_(passport);
  var exp = p.stamp ? new Date(p.stamp + PORTAL_CFG.SESSION_TTL_MS) : null;
  return {
    loginAt: p.stamp ? PORTAL_formatDateTime_(new Date(p.stamp)) : '',
    expiresAt: exp ? PORTAL_formatDateTime_(exp) : '',
    lastLogoutAt: PORTAL_getLastLogoutStamp_(email) ? PORTAL_formatDateTime_(new Date(PORTAL_getLastLogoutStamp_(email))) : '',
    ttlHours: Math.round(PORTAL_CFG.SESSION_TTL_MS / 3600000)
  };
}

function PORTAL_findUser_(email) {
  var sh = PORTAL_master_().getSheetByName(PORTAL_SHEETS.MASTER_USER);
  if (!sh) throw new Error('Sheet Master_User tidak ditemukan.');
  var rows = PORTAL_readRows_(sh);
  email = PORTAL_normEmail_(email);
  for (var i=0; i<rows.length; i++) {
    var e = PORTAL_pickValue_(rows[i], ['Email','Email_Google','Gmail','Email_User','User_Email']);
    if (PORTAL_normEmail_(e) === email) return rows[i];
  }
  return null;
}

function PORTAL_userDisplayName_(user, email) {
  return PORTAL_clean_(PORTAL_pickValue_(user || {}, ['Display_Name','Display Name','Nama','Nama User','Nama_User','Nama_Lengkap','Nama Lengkap','Name','User_Name','Username','User'])) || PORTAL_clean_(email);
}

function PORTAL_pickValue_(obj, aliases) {
  aliases = aliases || [];
  var map = {};
  Object.keys(obj || {}).forEach(function(k){ map[PORTAL_key_(k)] = obj[k]; });
  for (var i=0;i<aliases.length;i++) {
    var key = PORTAL_key_(aliases[i]);
    if (map[key] !== undefined) return map[key];
  }
  return '';
}

function PORTAL_ensureLogoutColumn_() {
  var sh = PORTAL_master_().getSheetByName(PORTAL_SHEETS.MASTER_USER);
  if (!sh) throw new Error('Master_User tidak ditemukan.');
  var lastCol = Math.max(1, sh.getLastColumn());
  var headers = sh.getRange(1,1,1,lastCol).getValues()[0].map(PORTAL_clean_);
  var map = PORTAL_headerMap_(headers);
  if (PORTAL_col_(map, ['Last_Logout_At','Global_Logout_At','Logout_At','Sesi_Terakhir_Logout'], false) === -1) {
    sh.getRange(1, lastCol + 1).setValue('Last_Logout_At');
  }
}

function PORTAL_markLogout_(email, stamp) {
  var sh = PORTAL_master_().getSheetByName(PORTAL_SHEETS.MASTER_USER);
  if (!sh) return;
  PORTAL_ensureLogoutColumn_();
  var vals = sh.getDataRange().getValues();
  if (vals.length < 2) return;
  var map = PORTAL_headerMap_(vals[0]);
  var cEmail = PORTAL_col_(map, ['Email','Email_Google','Gmail','Email_User','User_Email'], false);
  var cLogout = PORTAL_col_(map, ['Last_Logout_At','Global_Logout_At','Logout_At','Sesi_Terakhir_Logout'], false);
  if (cEmail < 0 || cLogout < 0) return;
  for (var r=1; r<vals.length; r++) {
    if (PORTAL_normEmail_(vals[r][cEmail]) === PORTAL_normEmail_(email)) {
      sh.getRange(r+1, cLogout+1).setValue(Number(stamp) || Date.now());
      return;
    }
  }
}

function PORTAL_getLastLogoutStamp_(email) {
  var user = PORTAL_findUser_(email);
  if (!user) return 0;
  var raw = PORTAL_pickValue_(user, ['Last_Logout_At','Global_Logout_At','Logout_At','Sesi_Terakhir_Logout']);
  if (!raw) return 0;
  if (raw instanceof Date) return raw.getTime();
  var n = Number(raw);
  if (n > 0) return n;
  var d = new Date(raw);
  return isNaN(d.getTime()) ? 0 : d.getTime();
}

function PORTAL_readGlobalHeartbeat_() {
  var sh = PORTAL_master_().getSheetByName(PORTAL_SHEETS.MASTER_MODULE);
  if (!sh) return { version:'0', updatedAt:'', notes:'Master_Module belum ada' };
  var version = String(sh.getRange(PORTAL_CFG.HEARTBEAT_CELL).getValue() || '0');
  var updated = sh.getRange(PORTAL_CFG.HEARTBEAT_UPDATED_CELL).getValue();
  var notes = sh.getRange(PORTAL_CFG.HEARTBEAT_NOTES_CELL).getValue();
  return { version:version, updatedAt:PORTAL_formatDateTime_(updated), notes:notes || '' };
}

function PORTAL_bumpHeartbeat_(notes) {
  var sh = PORTAL_master_().getSheetByName(PORTAL_SHEETS.MASTER_MODULE);
  if (!sh) return { version:'0', updatedAt:'', notes:'Master_Module belum ada' };
  var ts = Date.now();
  sh.getRange(PORTAL_CFG.HEARTBEAT_CELL).setValue(ts);
  sh.getRange(PORTAL_CFG.HEARTBEAT_UPDATED_CELL).setValue(new Date());
  sh.getRange(PORTAL_CFG.HEARTBEAT_NOTES_CELL).setValue(notes || 'Update');
  return PORTAL_readGlobalHeartbeat_();
}

function PORTAL_getPortalUrl_() {
  try {
    var rows = PORTAL_readRows_(PORTAL_master_().getSheetByName(PORTAL_SHEETS.MASTER_MODULE));
    for (var i=0;i<rows.length;i++) {
      var code=PORTAL_key_(rows[i].Module_Code || rows[i].Kode_Modul), name=PORTAL_key_(rows[i].Module_Name || rows[i].Nama_Modul);
      if (code==='PORTAL' || name.indexOf('PORTAL')!==-1 || name.indexOf('BERANDA')!==-1) return PORTAL_clean_(rows[i].Web_App_URL || rows[i].WebApp_URL || rows[i].URL_Web_Aktif || rows[i].URL || '');
    }
  } catch(e) {}
  return '';
}

  if (!s) return true;
  return ['OFF','DISABLED','INACTIVE','NONAKTIF','ARCHIVE','ARSIP','FALSE','STOP'].indexOf(s) === -1;
}
